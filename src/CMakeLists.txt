# The sources are in the directory LIB, or in the directory given as an optional fifth argument.
macro(__add_benchmark LIB LIB_CMAKE NAME TYPE)
  set(__SRC_DIR ${LIB})
  if (${ARGC} GREATER 4)
    set(__SRC_DIR ${ARGV4})
  endif()
  add_executable(${LIB}_${NAME}_${TYPE} ${__SRC_DIR}/${NAME}_${TYPE}.cpp)

  target_link_libraries(${LIB}_${NAME}_${TYPE} PRIVATE ${LIB_CMAKE})
  target_link_libraries(${LIB}_${NAME}_${TYPE} PRIVATE common)

  if (${LIB} STREQUAL "oxidd")
    set_property(TARGET ${LIB}_${NAME}_${TYPE} PROPERTY CXX_STANDARD 20)
  endif()

  if (BDD_BENCHMARK_GRENDEL)
    target_compile_definitions(${LIB}_${NAME}_${TYPE} PRIVATE BDD_BENCHMARK_GRENDEL)
  endif()
  if (BDD_BENCHMARK_INCL_INIT)
    target_compile_definitions(${LIB}_${NAME}_${TYPE} PRIVATE BDD_BENCHMARK_INCL_INIT)
  endif()
  if (BDD_BENCHMARK_STATS)
    target_compile_definitions(${LIB}_${NAME}_${TYPE} PRIVATE BDD_BENCHMARK_STATS)
  endif()
  if (BDD_BENCHMARK_WAIT)
    target_compile_definitions(${LIB}_${NAME}_${TYPE} PRIVATE BDD_BENCHMARK_WAIT)
  endif()

  # TODO: When all BDD packages support being used in a C++20 application, we can replace all lines
  #       below with this common one-liner.
  # set_target_properties(${LIB}_${NAME}_${TYPE} PROPERTIES C_STANDARD 20 CXX_STANDARD 20)
endmacro(__add_benchmark)

macro(add_bdd_benchmark NAME)
  if (ADIAR)
    __add_benchmark(adiar  adiar   ${NAME} bdd)
    set_target_properties(adiar_${NAME}_bdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (ADIAR)
  if (BUDDY)
    __add_benchmark(buddy  buddy   ${NAME} bdd)
    set_target_properties(buddy_${NAME}_bdd PROPERTIES C_STANDARD 11 CXX_STANDARD 11)
  endif (BUDDY)
  if (CUDD)
    __add_benchmark(cudd   cudd    ${NAME} bdd)
    set_target_properties(cudd_${NAME}_bdd PROPERTIES C_STANDARD 11 CXX_STANDARD 11)
  endif (CUDD)
  if (LIBBDD)
    __add_benchmark(libbdd lib-bdd ${NAME} bdd)
    set_target_properties(libbdd_${NAME}_bdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (LIBBDD)
  if (OXIDD)
    __add_benchmark(oxidd  oxidd   ${NAME} bdd)
    set_target_properties(oxidd_${NAME}_bdd PROPERTIES C_STANDARD 20 CXX_STANDARD 20)
  endif (OXIDD)
  if (RUDDY)
    __add_benchmark(ruddy  ruddy   ${NAME} bdd)
    set_target_properties(ruddy_${NAME}_bdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (RUDDY)
endmacro(add_bdd_benchmark)

macro(add_bcdd_benchmark NAME)

    set_target_properties(aytdd_${NAME}_bcdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (AYTDD)
  if (AYTDD_OPEN_TABLE)
    __add_benchmark(aytdd_ot aytdd_ot     ${NAME} bcdd aytdd)
    target_compile_definitions(aytdd_ot_${NAME}_bcdd PRIVATE "AYTDD_NAME=\"Aytdd open table\"")
    set_target_properties(aytdd_ot_${NAME}_bcdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (AYTDD_OPEN_TABLE)
  if (CAL)
    __add_benchmark(cal    cal            ${NAME} bcdd)
    set_target_properties(cal_${NAME}_bcdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (CAL)
  if (CUDD)
    __add_benchmark(cudd   cudd           ${NAME} bcdd)
    set_target_properties(cudd_${NAME}_bcdd PROPERTIES C_STANDARD 11 CXX_STANDARD 11)
  endif (CUDD)
  if (OXIDD)
    __add_benchmark(oxidd  oxidd          ${NAME} bcdd)
    set_target_properties(oxidd_${NAME}_bcdd PROPERTIES C_STANDARD 20 CXX_STANDARD 20)
  endif (OXIDD)
  if (RSDD)
    __add_benchmark(rsdd   rsdd           ${NAME} bcdd)
    set_target_properties(rsdd_${NAME}_bcdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (RSDD)
  if (SIMPLEABC)
    __add_benchmark(simpleabc libabc      ${NAME} bcdd)
    set_target_properties(simpleabc_${NAME}_bcdd PROPERTIES C_STANDARD 17 CXX_STANDARD 17)
  endif (SIMPLEABC)
  if (SYLVAN)

  endif (SYLVAN)
